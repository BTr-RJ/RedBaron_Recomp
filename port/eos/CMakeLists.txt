cmake_minimum_required(VERSION 3.25)
project(redbaron_eos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REDBARON_EOS_SDK "" CACHE PATH "Path to the official extracted EOS SDK directory")
if(NOT WIN32)
    if(NOT EXISTS "${REDBARON_EOS_SDK}/Bin/libEOSSDK-Linux-Shipping.so")
        message(FATAL_ERROR "The official Linux EOS SDK is required for the peer diagnostic")
    endif()
    add_executable(redbaron_eos_peer_check peer_check.cpp platform.cpp eos_wire.cpp fragmentation.cpp
        address_wire.cpp roster.cpp lobby_roster.cpp lobby_directory.cpp lobby_join.cpp lobby_record.cpp session_record.cpp ../network/session_record.cpp)
    target_include_directories(redbaron_eos_peer_check PRIVATE "${REDBARON_EOS_SDK}/Include")
    target_link_libraries(redbaron_eos_peer_check PRIVATE "${REDBARON_EOS_SDK}/Bin/libEOSSDK-Linux-Shipping.so" pthread dl)
    return() # Diagnostic only; this is not a Linux game port.
endif()
foreach(required Include/eos_sdk.h Lib/EOSSDK-Win64-Shipping.lib Bin/EOSSDK-Win64-Shipping.dll)
    if(NOT EXISTS "${REDBARON_EOS_SDK}/${required}")
        message(FATAL_ERROR "Official EOS SDK component missing: ${required}")
    endif()
endforeach()
add_executable(redbaron_eos_probe eos_probe.cpp ../network/session_publisher.cpp ../network/directory_responder.cpp
    ../network/selection_gateway.cpp ../network/selection_admission.cpp ../network/socket_bridge.cpp)
target_compile_definitions(redbaron_eos_probe PRIVATE REDBARON_STANDALONE_BRIDGE_TEST)
target_include_directories(redbaron_eos_probe PRIVATE "${REDBARON_EOS_SDK}/Include")
target_link_libraries(redbaron_eos_probe PRIVATE "${REDBARON_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib" ws2_32)
add_custom_command(TARGET redbaron_eos_probe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${REDBARON_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll" "$<TARGET_FILE_DIR:redbaron_eos_probe>")
enable_testing()
add_test(NAME redbaron_eos_sdk_lifecycle COMMAND redbaron_eos_probe --sdk-check)
add_executable(redbaron_eos_fragment_tests fragmentation.cpp fragmentation_test.cpp)

add_library(redbaron_eos_lobby STATIC session_record.cpp lobby_record.cpp lobby_publisher.cpp lobby_directory.cpp lobby_join.cpp join_selected.cpp selection_join.cpp directory_mapping.cpp roster.cpp lobby_roster.cpp ../network/session_record.cpp)

target_compile_definitions(redbaron_eos_wire_tests PRIVATE REDBARON_STANDALONE_BRIDGE_TEST)

        "${REDBARON_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll" "$<TARGET_FILE_DIR:redbaron_eos_wire_tests>")

add_test(NAME redbaron_eos_session_record COMMAND redbaron_eos_record_tests)

target_include_directories(redbaron_eos_directory_tests PRIVATE "${REDBARON_EOS_SDK}/Include")
target_link_libraries(redbaron_eos_directory_tests PRIVATE ws2_32)
add_test(NAME redbaron_eos_directory_mapping COMMAND redbaron_eos_directory_tests)

if(NOT TARGET rex::runtime AND rexglue_DIR)
    find_package(rexglue 0.10.0 REQUIRED CONFIG)

if(TARGET rex::runtime)
    add_library(redbaron_eos_service STATIC platform.cpp social.cpp achievement_sync.cpp address_wire.cpp native_record.cpp online_service.cpp ../network/socket_bridge.cpp)
    target_compile_features(redbaron_eos_service PUBLIC cxx_std_23)
    target_compile_definitions(redbaron_eos_service PRIVATE REDBARON_STANDALONE_BRIDGE_TEST)
    target_include_directories(redbaron_eos_service PRIVATE "${REDBARON_EOS_SDK}/Include")
    target_link_libraries(redbaron_eos_service PUBLIC rex::runtime PRIVATE redbaron_eos_wire redbaron_eos_lobby
        "${REDBARON_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib" ws2_32)
    add_executable(redbaron_eos_native_tests native_provider_test.cpp native_record.cpp ../network/session_record.cpp)
    target_compile_features(redbaron_eos_native_tests PRIVATE cxx_std_23)
    target_link_libraries(redbaron_eos_native_tests PRIVATE rex::runtime)
    add_custom_command(TARGET redbaron_eos_native_tests POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:rex::runtime>" "$<TARGET_FILE_DIR:redbaron_eos_native_tests>")
    add_test(NAME redbaron_eos_native_provider COMMAND redbaron_eos_native_tests)
    set_tests_properties(redbaron_eos_native_provider PROPERTIES TIMEOUT 20)
    add_executable(redbaron_eos_service_check service_check.cpp)
    target_include_directories(redbaron_eos_service_check PRIVATE "${REDBARON_EOS_SDK}/Include")
    target_link_libraries(redbaron_eos_service_check PRIVATE redbaron_eos_service redbaron_eos_lobby
        "${REDBARON_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib")
    get_filename_component(service_sdk_root "${rexglue_DIR}/../../.." ABSOLUTE)
    foreach(host redbaron_eos_native_tests redbaron_eos_service_check)
        foreach(asset rexruntime.dll amd_fidelityfx_upscaler_dx12.dll)
            if(EXISTS "${service_sdk_root}/bin/${asset}")
                add_custom_command(TARGET ${host} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${service_sdk_root}/bin/${asset}" "$<TARGET_FILE_DIR:${host}>")
            endif()
        endforeach()
    endforeach()
    add_custom_command(TARGET redbaron_eos_service_check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${REDBARON_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll" "$<TARGET_FILE_DIR:redbaron_eos_service_check>")
endif()
