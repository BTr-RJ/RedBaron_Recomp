cmake_minimum_required(VERSION 3.25)
project(redbaron_eos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REDBARON_EOS_SDK "" CACHE PATH "Path to the official extracted EOS SDK directory")
if(NOT WIN32)
    message(FATAL_ERROR "The EOS diagnostic currently supports Windows only")
endif()
foreach(required Include/eos_sdk.h Lib/EOSSDK-Win64-Shipping.lib Bin/EOSSDK-Win64-Shipping.dll)
    if(NOT EXISTS "${REDBARON_EOS_SDK}/${required}")
        message(FATAL_ERROR "Official EOS SDK component missing: ${required}")
    endif()
endforeach()
add_executable(redbaron_eos_probe eos_probe.cpp)
target_include_directories(redbaron_eos_probe PRIVATE "${REDBARON_EOS_SDK}/Include")
target_link_libraries(redbaron_eos_probe PRIVATE "${REDBARON_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib")
add_custom_command(TARGET redbaron_eos_probe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${REDBARON_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll" "$<TARGET_FILE_DIR:redbaron_eos_probe>")
enable_testing()
add_test(NAME redbaron_eos_sdk_lifecycle COMMAND redbaron_eos_probe --sdk-check)
add_executable(redbaron_eos_fragment_tests fragmentation.cpp fragmentation_test.cpp)
target_include_directories(redbaron_eos_fragment_tests PRIVATE "${REDBARON_EOS_SDK}/Include")
add_test(NAME redbaron_eos_fragmentation COMMAND redbaron_eos_fragment_tests)
