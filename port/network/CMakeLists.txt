cmake_minimum_required(VERSION 3.25)
project(redbaron_net LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REDBARON_DATACHANNEL_SOURCE "" CACHE PATH "Pinned libdatachannel source with dependencies")
set(REDBARON_JSON_INCLUDE "" CACHE PATH "Pinned nlohmann single_include directory")
if(NOT EXISTS "${REDBARON_DATACHANNEL_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Prepare the pinned networking dependencies first")
endif()
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${REDBARON_DATACHANNEL_SOURCE}" datachannel)
target_sources(datachannel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/crypto_threading.cpp")
add_executable(redbaron_net main.cpp transport.cpp socket_bridge.cpp signaling.cpp directory.cpp session_publisher.cpp session_record.cpp directory_responder.cpp selection_gateway.cpp)
target_include_directories(redbaron_net PRIVATE "${REDBARON_JSON_INCLUDE}")
target_link_libraries(redbaron_net PRIVATE datachannel ws2_32 winhttp)
target_compile_definitions(redbaron_net PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_custom_command(TARGET redbaron_net POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:datachannel> $<TARGET_FILE_DIR:redbaron_net>)
enable_testing()
add_executable(redbaron_directory_responder_tests directory_responder_test.cpp directory_responder.cpp session_record.cpp)
target_link_libraries(redbaron_directory_responder_tests PRIVATE ws2_32)
add_test(NAME redbaron_directory_responder COMMAND redbaron_directory_responder_tests)
add_executable(redbaron_selection_gateway_tests selection_gateway_test.cpp selection_gateway.cpp socket_bridge.cpp)
target_compile_definitions(redbaron_selection_gateway_tests PRIVATE REDBARON_STANDALONE_BRIDGE_TEST)
target_link_libraries(redbaron_selection_gateway_tests PRIVATE ws2_32)
add_test(NAME redbaron_selection_gateway COMMAND redbaron_selection_gateway_tests)
